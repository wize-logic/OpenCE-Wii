# A two-player co-op campaign start of a10, for debug.test_input =
# "script:coop.txt" (port/linux/src/test_input.c lists the steps;
# port/wii/README.md, "The playthrough check"), with the Xbox's menus
# (display.menus = "xbox", the Wii's only ones) and an empty saves folder.
# Controller 2 is made to appear with "connect 2" (the Xbox's device insertion:
# XGetDeviceChanges and XInputOpen; on the Wii xinput_wii.c's port_connected),
# so the run needs no second Remote or pad. Every line the log gets starts
# "test_input:"; each "step" below is one of the checkpoint's steps.
#
# MULTIPLAYER, CO-OP CAMPAIGN: player 1's profile (A of controller 1), player
# 2's (A of controller 2), the level (a10, the first in the list) and the
# difficulty (NORMAL, either controller). Two players then stand in the cryo
# bay; the "local player" lines of the x86 build's log, and the game's own
# ("player count 2"), say both are there.

# the main menu: a profile first (CAMPAIGN makes it), back to the main menu
screen main_menu\main_menu 300
wait 2
log step 1: the first profile (CAMPAIGN makes it), back to the main menu
press a
screen new_campaign_creating_profile 60
wait 2
press a
screen difficulty_select 120
wait 2
press b
wait 3
screen main_menu\main_menu 60

# MULTIPLAYER, CO-OP CAMPAIGN
log step 2: MULTIPLAYER, CO-OP CAMPAIGN
select main_menu_item_multiplayer
press a
wait 2
select multiplayer_type_coop_item
press a
screen player1_profile_select 60
wait 2
connect 2
wait 2

# the two profiles (the first New001 is both players'), the level, the difficulty
log step 3: player 1's profile, player 2's
press a
screen player2_profile_select 60
wait 2
2: press a
screen solo_level_select 60
wait 2
log step 4: the level (a10), the difficulty (NORMAL)
press a
screen difficulty_select 60
wait 2
select normal_difficulty_item
press a

# a10 loads; its opening cinematic is skipped (A), which goes back to the
# checkpoint saved at its first tick
play 300
log step 5: a10 plays, both players in the cryo bay
wait 8
log step 6: the opening cinematic skipped
press a
ticks 150

# the cryo tubes: the look training of each player, then X held until each is
# out of its tube
log step 7: the look training, player 1's and player 2's
repeat 6
stick 0 0 -1 0 1.5
stick 0 0 1 0 1.5
stick 0 0 0 1 1.5
stick 0 0 0 -1 1.5
press a
end
repeat 6
2: stick 0 0 -1 0 1.5
2: stick 0 0 1 0 1.5
2: stick 0 0 0 1 1.5
2: stick 0 0 0 -1 1.5
2: press a
end
stick 0 0 0 1 1
2: stick 0 0 0 1 1
log step 8: out of the cryo tubes
repeat 12
hold x 5 afoot
wait 1
end
afoot 30
2: hold x 5
log step 9: both players afoot
ticks 90

# the bots play both for 30 seconds
bot 1 1
bot 2 2
repeat 3
ticks 300
log step 10: both players play
end
quit
